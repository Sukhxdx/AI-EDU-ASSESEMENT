from transformers import AutoTokenizer, AutoModelForSeq2SeqLM, pipeline

class QuizGenerator:
    def __init__(self, model_name='t5-small'):
        self.pipe = pipeline('text2text-generation', model=model_name, tokenizer=model_name)

    def generate(self, text, num_questions=5):
        from nltk.tokenize import sent_tokenize
        sents = sent_tokenize(text)
        sents = [s for s in sents if len(s.split()) > 6]
        sents = sorted(sents, key=len, reverse=True)
        questions = []
        for sent in sents[:num_questions]:
            prompt = f"generate question: {sent}"
            q = self.pipe(prompt, max_length=64, num_return_sequences=1)[0]['generated_text']
            questions.append({'question': q})
        return questions
