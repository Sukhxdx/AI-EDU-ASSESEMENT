import sqlite3, json
from datetime import datetime

class Database:
    def __init__(self, path='data/projects.db'):
        self.conn = sqlite3.connect(path, check_same_thread=False)
        self._init_db()

    def _init_db(self):
        cur = self.conn.cursor()
        cur.execute('''CREATE TABLE IF NOT EXISTS sessions(
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            created_at TEXT,
            text TEXT,
            quizzes TEXT,
            readability TEXT,
            glossary TEXT
        )''')
        self.conn.commit()

    def save_session(self, text, quizzes, readability, glossary):
        cur = self.conn.cursor()
        cur.execute('INSERT INTO sessions(created_at, text, quizzes, readability, glossary) VALUES(?,?,?,?,?)',
                    (datetime.utcnow().isoformat(), text, json.dumps(quizzes), json.dumps(readability), json.dumps(glossary)))
        self.conn.commit()
        return cur.lastrowid

    def list_sessions(self):
        cur = self.conn.cursor()
        cur.execute('SELECT id, created_at FROM sessions ORDER BY id DESC')
        return [{'id': row[0], 'created_at': row[1]} for row in cur.fetchall()]

    def load_session(self, sid):
        cur = self.conn.cursor()
        cur.execute('SELECT text, quizzes, readability, glossary FROM sessions WHERE id=?', (sid,))
        row = cur.fetchone()
        if not row:
            return None
        return {'text': row[0], 'quizzes': json.loads(row[1]), 'readability': json.loads(row[2]), 'glossary': json.loads(row[3])}
